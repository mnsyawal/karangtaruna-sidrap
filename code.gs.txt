function handleRequest(e, method) {
  try {
    const action = e.parameter.action;
    let payload = {};

    if (method === 'POST' && e.postData && e.postData.contents) {
      payload = JSON.parse(e.postData.contents);
    }

    if (method === 'GET') {
      switch (action) {
        case 'getBerita': return getBerita(e);
        case 'getBeritaById': return getBeritaById(e);
        case 'getArsip': return getArsip(e);
        case 'getStruktur': return getStruktur(e);
        case 'getKegiatan': return getKegiatan(e);
        case 'getDokumentasi': return getDokumentasi(e);
        case 'getAdminDokumentasi': return requireAuth(e, ['ADMIN', 'USER'], getAdminDokumentasi);
        case 'getAdminStruktur': return requireAuth(e, ['ADMIN'], getAdminStruktur);
        case 'getAdminKegiatan': return requireAuth(e, ['ADMIN', 'USER'], getAdminKegiatan);
        case 'getUsers': return requireAuth(e, ['ADMIN'], getUsers);
        case 'getStatistik': return getStatistik(e);
        case 'getSettings': return getSettings(e);
        default: return respondJSON(false, 'Action GET tidak valid.', null, null, 'INVALID_ACTION');
      }
    } 
    
    else if (method === 'POST') {
      const pAction = payload.action || action;
      const data = payload.data;

      switch (pAction) {
        case 'login': return login(data);
        
        case 'createBerita': return requireAuth(payload, ['ADMIN', 'USER'], () => createBerita(data, payload.user));
        case 'updateBerita': return requireAuth(payload, ['ADMIN', 'USER'], () => updateBerita(data, payload.user));
        case 'deleteBerita': return requireAuth(payload, ['ADMIN'], () => deleteBerita(data, payload.user));
        
        case 'createArsip': return requireAuth(payload, ['ADMIN', 'USER'], () => createArsip(data, payload.user));
        case 'updateArsip': return requireAuth(payload, ['ADMIN', 'USER'], () => updateArsip(data, payload.user));
        case 'deleteArsip': return requireAuth(payload, ['ADMIN'], () => deleteArsip(data, payload.user));
        
        case 'createDokumentasi': return requireAuth(payload, ['ADMIN', 'USER'], () => createDokumentasi(data, payload.user));
        case 'updateDokumentasi': return requireAuth(payload, ['ADMIN', 'USER'], () => updateDokumentasi(data, payload.user));
        case 'deleteDokumentasi': return requireAuth(payload, ['ADMIN'], () => deleteDokumentasi(data, payload.user));

        case 'createKegiatan': return requireAuth(payload, ['ADMIN', 'USER'], () => createKegiatan(data, payload.user));
        case 'updateKegiatan': return requireAuth(payload, ['ADMIN', 'USER'], () => updateKegiatan(data, payload.user));
        case 'deleteKegiatan': return requireAuth(payload, ['ADMIN'], () => deleteKegiatan(data, payload.user));

function login(data) {
  if (!data || !data.username || !data.password) {
    return respondJSON(false, 'Username dan password wajib diisi.', null, null, 'VALIDATION_ERROR');
  }
  
  const users = getSheetData('USERS');
  const hashedInputPassword = hashSHA256(data.password);
  
  const user = users.find(u => u.username === data.username && u.password_hash === hashedInputPassword && (u.status || '').toLowerCase() === 'active');
  
  if (!user) {
    return respondJSON(false, 'Username atau password salah (atau akun nonaktif).', null, null, 'AUTH_FAILED');
  }
  
  const token = generateToken(user);
  insertLog(user.id, user.username, 'LOGIN', 'AUTH', user.id);
  
  return respondJSON(true, 'Login berhasil.', {
    token: token,
    user: { id: user.id, username: user.username, nama: user.nama, role: user.role }
  });
}

function generateToken(user) {
  const timestamp = new Date().getTime();
  const rawSignature = user.id + '.' + user.role + '.' + timestamp + '.' + CONFIG.SECRET_KEY;
  const signature = hashSHA256(rawSignature);
  return user.id + '|' + user.role + '|' + timestamp + '|' + signature;
}

function verifyToken(tokenStr) {
  if (!tokenStr) return null;
  const parts = tokenStr.split('|');
  if (parts.length !== 4) return null;
  
  const [id, role, timestamp, signature] = parts;
  const now = new Date().getTime();
  if (now - parseInt(timestamp) > CONFIG.TOKEN_EXPIRY) return null;
  
  const expectedRaw = id + '.' + role + '.' + timestamp + '.' + CONFIG.SECRET_KEY;
  if (signature === hashSHA256(expectedRaw)) {
    return { id, role, username: 'user_auth' }; 
  }
  return null;
}

function requireAuth(requestObj, allowedRoles, callback) {
  const token = requestObj.parameter ? requestObj.parameter.token : requestObj.token;
  const user = verifyToken(token);
  
  if (!user) return respondJSON(false, 'Akses ditolak. Token tidak valid/kadaluarsa.', null, null, 'UNAUTHORIZED');
  if (!allowedRoles.includes(user.role)) return respondJSON(false, 'Akses ditolak. Anda tidak memiliki izin.', null, null, 'FORBIDDEN');
  
  requestObj.user = user;
  return callback();
}

function hashSHA256(input) {
  const rawHash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, input);
  let txtHash = '';
  for (let i = 0; i < rawHash.length; i++) {
    let hashVal = rawHash[i];
    if (hashVal < 0) { hashVal += 256; }
    if (hashVal.toString(16).length == 1) { txtHash += '0'; }
    txtHash += hashVal.toString(16);
  }
  return txtHash;
}

function getBerita(e) {

  const token = e.parameter.token;
  const user = verifyToken(token);
  const isPublic = !user;
  
  let data = getSheetData('BERITA');

    data = data.filter(item => (item.status || '').toString().trim().toLowerCase() === 'published');
  }
  if (category) data = data.filter(item => item.kategori === category);

  data.sort((a, b) => (new Date(b.created_at).getTime() || 0) - (new Date(a.created_at).getTime() || 0));
  
  const paginated = paginateData(data, page, limit);

}

function generateCleanRingkasan(isiHtml, explicitRingkasan, judul) {
  if (explicitRingkasan && explicitRingkasan.trim() !== '') {
    return explicitRingkasan.trim();
  }
  if (!isiHtml) return 'Klik untuk membaca selengkapnya.';
  var cleaned = isiHtml
    .replace(/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi, ' ')
    .replace(/<style\b[^<]*(?:(?!<\/style>)<[^<]*)*<\/style>/gi, ' ')
    .replace(/<iframe\b[^<]*(?:(?!<\/iframe>)<[^<]*)*<\/iframe>/gi, ' ')
    .replace(/<[^>]+>/g, ' ')
    .replace(/&nbsp;/gi, ' ')
    .replace(/&lt;[^&]*&gt;/gi, ' ')
    .replace(/\s+/g, ' ')
    .trim();
  if (!cleaned || cleaned.length < 5) {
    return judul ? ('Informasi selengkapnya mengenai "' + judul + '". Klik untuk membaca detail.') : 'Klik untuk membaca selengkapnya.';
  }
  return cleaned.length > 150 ? (cleaned.substring(0, 150).trim() + '...') : cleaned;
}

function updateSettings(data, user) {
  const sheet = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID).getSheetByName('SETTINGS');
  const sheetData = sheet.getDataRange().getValues();
  
  if (data.logoBase64) {
    const logoUrl = uploadFile(data.logoBase64, 'LOGO_WEB_' + new Date().getTime(), CONFIG.FOLDER_DOKUMENTASI_ID);
    data.settings.logo_url = logoUrl; 
  }

  if (data.heroBgBase64) {
    const heroUrl = uploadFile(data.heroBgBase64, 'HERO_BG_' + new Date().getTime(), CONFIG.FOLDER_DOKUMENTASI_ID);
    data.settings.hero_bg_image = heroUrl;
  }

  for (let key in data.settings) {
    let keyFound = false;
    for (let i = 1; i < sheetData.length; i++) {
      if (sheetData[i][0] === key) {
        sheet.getRange(i + 1, 2).setValue(data.settings[key]);
        keyFound = true;
        break;
      }
    }
    if (!keyFound) {
      sheet.appendRow([key, data.settings[key], '']);
    }
  }
  
  insertLog(user.id, user.username, 'UPDATE', 'SETTINGS', 'Global_Settings');
  return respondJSON(true, 'Pengaturan web berhasil disimpan.', data.settings);
}

    const imageUrl = uploadFile(data.imageBase64, filename, CONFIG.FOLDER_BERITA_ID);
    
    if (!imageUrl) {
      return respondJSON(false, 'Gagal mengunggah gambar ke Drive.');
    }
    
    insertLog(user.id, user.username, 'UPLOAD_IMAGE', 'BERITA', filename);
    
    return respondJSON(true, 'Gambar berhasil diunggah.', { url: imageUrl });
  } catch (e) {
    return respondJSON(false, 'Terjadi kesalahan saat mengunggah gambar.');
  }
}

function getSheetData(sheetName) {
  const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    if (sheetName === 'KEGIATAN') {
      sheet = ss.insertSheet('KEGIATAN');
      sheet.appendRow(['id', 'waktu', 'tanggal', 'jenis_kegiatan', 'keterangan', 'pic', 'lokasi', 'status', 'created_by', 'updated_by', 'created_at', 'updated_at']);
      return [];
    }
    return [];
  }
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  const headers = data[0];
  const result = [];
  
  for (let i = 1; i < data.length; i++) {
    if (!data[i][0] || data[i][0].toString().trim() === '') continue; 
    
    let obj = {};
    for (let j = 0; j < headers.length; j++) {
      obj[headers[j]] = data[i][j];
    }
    result.push(obj);
  }
  return result;
}

function insertRow(sheetName, dataObj) {
  const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet && sheetName === 'KEGIATAN') {
    sheet = ss.insertSheet('KEGIATAN');
    sheet.appendRow(['id', 'waktu', 'tanggal', 'jenis_kegiatan', 'keterangan', 'pic', 'lokasi', 'status', 'created_by', 'updated_by', 'created_at', 'updated_at']);
  }
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const rowData = [];
  for (let i = 0; i < headers.length; i++) {
    rowData.push(dataObj[headers[i]] !== undefined ? dataObj[headers[i]] : '');
  }
  sheet.appendRow(rowData);
}

function updateRow(sheetName, id, dataObj) {
  const sheet = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID).getSheetByName(sheetName);
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === id) { 
      for (let key in dataObj) {
        let colIndex = headers.indexOf(key);
        if (colIndex > -1 && dataObj[key] !== undefined) {
          sheet.getRange(i + 1, colIndex + 1).setValue(dataObj[key]);
        }
      }
      break;
    }
  }
}

function deleteRow(sheetName, id) {
  const sheet = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID).getSheetByName(sheetName);
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === id) {
      sheet.deleteRow(i + 1);
      break;
    }
  }
}

function uploadFile(base64Data, filename, folderId) {
  try {
    const splitBase = base64Data.split(',');
    if (splitBase.length < 2) {
      console.error('[UPLOAD] base64Data tidak valid (tidak ada koma pemisah)');
      return '';
    }

    const mimeMatch = splitBase[0].match(/data:([^;]+)/);
    const mimeType = mimeMatch ? mimeMatch[1] : 'application/octet-stream';
    const byteCharacters = Utilities.base64Decode(splitBase[1]);
    const blob = Utilities.newBlob(byteCharacters, mimeType, filename);
    const folder = DriveApp.getFolderById(folderId);
    
    const file = folder.createFile(blob);
    console.log('[UPLOAD] File dibuat: ' + file.getName() + ' | ID: ' + file.getId());
    
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    
    const viewers = file.getViewers();
    const editors = file.getEditors();
    
    const access = file.getSharingAccess();
    const permission = file.getSharingPermission();
    console.log('[UPLOAD] Sharing access: ' + access + ' | Permission: ' + permission);
    
    const fileId = file.getId();
    
    let url;
    if (mimeType.startsWith('image/')) {
      url = 'https://drive.google.com/thumbnail?id=' + fileId + '&sz=w1000';
    } else if (mimeType === 'application/pdf') {
      url = 'https://drive.google.com/file/d/' + fileId + '/preview';
    } else {
      url = 'https://drive.google.com/uc?export=download&id=' + fileId;
    }
    
    console.log('[UPLOAD] URL publik: ' + url);
    return url;
    
  } catch (e) {
    console.error('[UPLOAD ERROR] ' + e.toString());
    return '';
  }
}

function respondJSON(success, message, data = null, pagination = null, errorCode = null) {
  const res = { success, message };
  if (data) res.data = data;
  if (pagination) res.pagination = pagination;
  if (errorCode) res.error_code = errorCode;
  return ContentService.createTextOutput(JSON.stringify(res)).setMimeType(ContentService.MimeType.JSON);
}

function paginateData(array, page, limit) {
  const startIndex = (page - 1) * limit;
  const endIndex = page * limit;
  return {
    data: array.slice(startIndex, endIndex),
    meta: { page: page, limit: limit, total: array.length, total_pages: Math.ceil(array.length / limit) }
  };
}

function generateSlug(text) {
  return text.toString().toLowerCase().replace(/\s+/g, '-').replace(/[^\w\-]+/g, '').replace(/\-\-+/g, '-').replace(/^-+/, '').replace(/-+$/, '');
}
